    <script>

        // --- ELEMENTOS DEL DOM ---

        const initialInvestmentEl = document.getElementById('initialInvestment');

        const contributionAmountEl = document.getElementById('contributionAmount');

        const contributionFrequencyEl = document.getElementById('contributionFrequency');

        const investmentYearsEl = document.getElementById('investmentYears');

        const yearsValueEl = document.getElementById('yearsValue');

        const interestRateValueEl = document.getElementById('interestRateValue');

        const futureValueEl = document.getElementById('futureValue');

        const totalContributionsValueEl = document.getElementById('totalContributionsValue');

        const totalEarningsValueEl = document.getElementById('totalEarningsValue');

        const chartCanvas = document.getElementById('savingsChart');


        // --- MAPA DE TASAS DE INTERÉS ---

        // (Año: Tasa mensual en porcentaje)

        const interestRates = {

            1: 1.6, 2: 1.7, 3: 1.8, 4: 1.9, 5: 2.0, 6: 2.1, 7: 2.2,

            8: 2.3, 9: 2.4, 10: 2.5, 11: 2.6, 12: 2.7, 13: 2.8, 14: 2.9, 15: 3.0

        };


        let savingsChart; // Variable para mantener la instancia del gráfico


        // --- FUNCIÓN PRINCIPAL DE CÁLCULO ---

        function calculateProjection() {

            // 1. Obtener los valores de los inputs

            const initialInvestment = parseFloat(initialInvestmentEl.value) || 0;

            const contributionAmount = parseFloat(contributionAmountEl.value) || 0;

            const frequency = contributionFrequencyEl.value;

            const years = parseInt(investmentYearsEl.value);


            // 2. Obtener la tasa de interés mensual

            const monthlyInterestRate = interestRates[years] / 100;


            // 3. Simulación mes a mes

            let balance = initialInvestment;

            let totalContributions = initialInvestment; // Incluye la inversión inicial

            const totalMonths = years * 12;


            const chartLabels = [];

            const chartTotalData = [];

            const chartContributionData = [];


            for (let month = 1; month <= totalMonths; month++) {

                // Añadir aportación según la frecuencia

                switch(frequency) {

                    case 'semanal':

                        // Se promedia 4.33 semanas por mes

                        balance += contributionAmount * (52 / 12);

                        totalContributions += contributionAmount * (52/12);

                        break;

                    case 'mensual':

                        balance += contributionAmount;

                        totalContributions += contributionAmount;

                        break;

                    case 'semestral':

                        if (month % 6 === 0) {

                            balance += contributionAmount;

                            totalContributions += contributionAmount;

                        }

                        break;

                    case 'anual':

                         if (month % 12 === 0) {

                            balance += contributionAmount;

                            totalContributions += contributionAmount;

                        }

                        break;

                }

                

                // Aplicar interés compuesto mensual

                balance *= (1 + monthlyInterestRate);


                // Guardar datos para el gráfico al final de cada año

                if (month % 12 === 0) {

                    chartLabels.push(`Año ${month / 12}`);

                    chartTotalData.push(balance.toFixed(2));

                    chartContributionData.push(totalContributions.toFixed(2));

                }

            }

             

            // Si el plazo no es un número entero de años, se añade el último punto

            if (totalMonths % 12 !== 0) {

                chartLabels.push(`Año ${years}`);

                chartTotalData.push(balance.toFixed(2));

                chartContributionData.push(totalContributions.toFixed(2));

            }


            // 4. Actualizar la UI

            const totalEarnings = balance - totalContributions;

            updateUI(years, balance, totalContributions, totalEarnings, chartLabels, chartTotalData, chartContributionData);

        }


        // --- FUNCIÓN PARA ACTUALIZAR LA INTERFAZ ---

        function updateUI(years, finalBalance, totalContributions, totalEarnings, chartLabels, chartTotalData, chartContributionData) {

            // Actualizar textos

            yearsValueEl.textContent = `${years} Año${years > 1 ? 's' : ''}`;

            interestRateValueEl.textContent = `${interestRates[years]}%`;

            futureValueEl.textContent = finalBalance.toLocaleString('es-MX', { style: 'currency', currency: 'MXN' });

            totalContributionsValueEl.textContent = totalContributions.toLocaleString('es-MX', { style: 'currency', currency: 'MXN' });

            totalEarningsValueEl.textContent = totalEarnings.toLocaleString('es-MX', { style: 'currency', currency: 'MXN' });

            

            // Actualizar el gráfico

            updateChart(chartLabels, chartTotalData, chartContributionData);

        }


        // --- FUNCIÓN PARA ACTUALIZAR EL GRÁFICO ---

        function updateChart(labels, totalData, contributionData) {

            if (savingsChart) {

                savingsChart.destroy();

            }


            // Calcular el rendimiento (ganancias)

            const returnsData = totalData.map((total, index) => total - contributionData[index]);


            const data = {

                labels: labels,

                datasets: [

                    {

                        label: 'Aportaciones',

                        data: contributionData,

                        backgroundColor: '#9CA3AF', // gray-400

                        borderColor: '#6B7280',    // gray-500

                        borderWidth: 1

                    },

                    {

                        label: 'Rendimiento',

                        data: returnsData,

                        backgroundColor: '#fcd34d', // amber-300

                        borderColor: '#fbbf24',    // amber-400

                        borderWidth: 1

                    }

                ]

            };


            const config = {

                type: 'bar',

                data: data,

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    plugins: {

                        legend: {

                            position: 'top',

                             labels: {

                                color: '#A0AEC0' // gray-400

                            }

                        },

                        tooltip: {

                            callbacks: {

                                label: function(context) {

                                    let label = context.dataset.label || '';

                                    if (label) {

                                        label += ': ';

                                    }

                                    if (context.parsed.y !== null) {

                                        label += new Intl.NumberFormat('es-MX', { style: 'currency', currency: 'MXN' }).format(context.parsed.y);

                                    }

                                    return label;

                                }

                            }

                        }

                    },

                    scales: {

                        x: {

                            stacked: true,

                             ticks: { color: '#A0AEC0' }, // gray-400

                             grid: { color: 'rgba(255, 255, 255, 0.1)' }

                        },

                        y: {

                            stacked: true,

                            beginAtZero: true,

                            ticks: { 

                                color: '#A0AEC0', // gray-400

                                callback: function(value) {

                                    return '$' + (value / 1000) + 'k';

                                }

                            },

                            grid: { color: 'rgba(255, 255, 255, 0.1)' }

                        }

                    }

                }

            };

            

            savingsChart = new Chart(chartCanvas, config);

        }



        // --- EVENT LISTENERS ---

        [initialInvestmentEl, contributionAmountEl, contributionFrequencyEl, investmentYearsEl].forEach(el => {

            el.addEventListener('input', calculateProjection);

        });


        // --- INICIALIZACIÓN ---

        document.addEventListener('DOMContentLoaded', calculateProjection);

    </script>
